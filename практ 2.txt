#include <iostream> (подключение ввода и вывода)
using namespace std; (строчка, позволяющая использовать функции из стандартных библиотек, не подписывая каждый раз std)

long long fact(int x) { (объявление функции, именно long long, а не int, потому что 13!уже не влезает в 32-битный int)
    long long r = 1; (локальная переменная накопитель)
    for (int i = 2; i <= x; i++) (счетчик)
        r *= i; (считаем факториал числа)
    return r; (возвращаем переменную)
}

int main() { (главная функция)
    int n; (объявили переменную)
    cout << "n = "; (выводим ее)
    cin >> n; (считывание числа)

    long long f = 1; (Накопитель произведения.)
    for (int i = 1; i <= n; i++) (Главный цикл: i пробегает 1…n)
        f *= fact(i); (и на каждом шаге f домножается на i факториал))

    cout << f << endl; (вывод результата)
    return 0; (завершение функции)
}
